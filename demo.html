Use it

<div id="face" style="width:160px;height:160px"></div>
<script src="agent-face.js"></script>
<script>
  const face = new AgentFace('#face', {
    shape: 'Pebble', color: 'blue', state: 'Idle'
  });

  face.setState('Attentive');   // user is typing
  face.setState('Busy');        // thinking
  face.setState('Excited');     // done
  face.setShape('Heart');       // morph the body
  face.setColor('orange');
</script>
